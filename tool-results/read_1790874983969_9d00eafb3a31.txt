    48→function EnvironmentBadge({ environment }: { environment: ToolEntry['environment'] }) {
    49→  if (environment === 'virtual-env') {
    50→    return (
    51→      <TooltipProvider delayDuration={150}>
    52→        <Tooltip>
    53→          <TooltipTrigger asChild>
    54→            <Badge variant="outline" className="border-amber-400/30 bg-amber-400/10 font-mono text-[10px] text-amber-300">virtual environment</Badge>
    55→          </TooltipTrigger>
    56→          <TooltipContent>Runs against the simulated server fleet</TooltipContent>
    57→        </Tooltip>
    58→      </TooltipProvider>
    59→    );
    60→  }
    61→  if (environment === 'js-function') {
    62→    return <Badge variant="outline" className="border-cyan-400/30 bg-cyan-400/10 font-mono text-[10px] text-cyan-300">js-function</Badge>;
    63→  }
    64→  if (environment === 'dynamic') {
    65→    return <Badge variant="outline" className="border-sky-400/30 bg-sky-400/10 font-mono text-[10px] text-sky-300">dynamic</Badge>;
    66→  }
    67→  return <Badge variant="outline" className="border-white/[0.09] font-mono text-[10px] text-muted-foreground">builtin</Badge>;

   224→  const [testTarget, setTestTarget] = useState<ToolEntry | null>(null);
   225→  const [testParams, setTestParams] = useState('{}');

   227→  const [testResult, setTestResult] = useState<ToolTestResult | null>(null);

   229→  const [deleteCandidate, setDeleteCandidate] = useState<ToolEntry | null>(null);

   246→      setError(null);

   252→  useEffect(() => {
   253→    void load();
   254→  }, [load]);
   255→
   256→  const onToggle = async (name: string, enabled: boolean) => {
   257→    setToggling(name);
   258→    try {
   259→      const updated = await toggleTool(name, enabled);
   260→      setTools((prev) => (prev ?? []).map((t) => (t.name === name ? { ...t, enabled: updated.enabled } : t)));
   261→      toast.success(`${name} ${updated.enabled ? 'enabled' : 'disabled'}`);
   262→    } catch (e) {
   263→      toast.error('Toggle failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });

   265→      setToggling(null);

   269→  const submitRegister = async () => {
   270→    if (!/^[a-z0-9_-]+\.[a-z0-9_-]+$/i.test(form.name.trim())) {
   271→      toast.error('Invalid tool name', { description: 'Use namespace.action, e.g. utility.summarize' });

   273→    }
   274→    let properties: unknown;
   275→    try {
   276→      properties = JSON.parse(form.schema);
   277→    } catch {
   278→      toast.error('Schema is not valid JSON');
   279→      return;
   280→    }
   281→    let handlerConfig: Record<string, unknown> | undefined;
   282→    if (form.handlerConfig.trim()) {
   283→      try {
   284→        handlerConfig = JSON.parse(form.handlerConfig);
   285→      } catch {
   286→        toast.error('handlerConfig is not valid JSON');
   287→        return;
   288→      }
   289→    }
   290→    setRegistering(true);
   291→    try {
   292→      await registerTool({
   293→        definition: {
   294→          name: form.name.trim(),
   295→          description: form.description.trim() || 'Dynamically registered tool',
   296→          category: form.category.trim() || 'utility',
   297→          ...(form.purpose.trim() ? { purpose: form.purpose.trim() } : {}),
   298→          environment: 'dynamic',
   299→          schema: { type: 'object', properties: Array.isArray(properties) ? properties : [] },
   300→        },
   301→        handlerKind: form.handlerKind as 'echo' | 'delay' | 'http_get' | 'uuid',
   302→        ...(handlerConfig ? { handlerConfig } : {}),
   303→      });
   304→      toast.success('Tool registered', { description: `${form.name} is now available to the CoreModule.` });
   305→      setRegOpen(false);
   306→      void load();
   307→    } catch (e) {
   308→      toast.error('Registration rejected', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });

   310→      setRegistering(false);
   311→    }
   312→  };

   382→  const onImportFile = async (file: File | undefined) => {
   383→    if (!file) return;
   384→    let text: string;
   385→    try {
   386→      text = await file.text();

   388→      toast.error('Import failed', { description: 'The file could not be read.' });
   389→      return;
   390→    }

   392→    if (!parsed.ok) {
   393→      setImportErrors({ errors: [parsed.error], warnings: [] });
   394→      return;
   395→    }
   396→    const result = validateImportedTool(parsed.value);
   397→    if (!result.ok || !result.tool) {
   398→      setImportErrors({ errors: result.errors, warnings: result.warnings });
   399→      return;
   400→    }
   401→    const existing = (tools ?? []).find((t) => t.name === result.tool!.name);
   402→    if (existing) {
   403→      setImportConflict({ tool: result.tool, existing });
   404→    } else {
   405→      setImportPreview({ tool: result.tool, warnings: result.warnings });
   406→    }

   427→            name: tool.name,
   428→            description: tool.description,
   429→            category: tool.category,
   430→            ...(tool.purpose ? { purpose: tool.purpose } : {}),
   431→            environment: 'dynamic',
   432→            schema: tool.schema,
   433→          },
   434→          handlerKind: tool.handlerKind as 'echo' | 'delay' | 'http_get' | 'uuid',

   438→      toast.success('Tool imported', { description: successDesc });
   439→      setImportPreview(null);
   440→      setImportConflict(null);

   469→      setImportBusy(false);
