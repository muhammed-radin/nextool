   112→export default function TaskPreviewView({ taskId }: { taskId: string }) {
   113→  const setActiveView = useConsoleStore((s) => s.setActiveView);
   114→
   115→  const [detail, setDetail] = useState<TaskDetail | null>(null);
   116→  const [detailError, setDetailError] = useState<string | null>(null);
   117→  const [executions, setExecutions] = useState<ToolExecution[] | null>(null);
   118→  const [context, setContext] = useState<ContextComposition | null>(null);
   119→  const [backfill, setBackfill] = useState<NexToolEvent[]>([]);
   120→
   121→  const [stopOpen, setStopOpen] = useState(false);
   122→  const [stopping, setStopping] = useState(false);
   123→  const [eventOpen, setEventOpen] = useState(false);
   124→  const [eventType, setEventType] = useState('user.message');
   125→  const [eventPayload, setEventPayload] = useState('{}');
   126→  const [feedbackOpen, setFeedbackOpen] = useState(false);
   127→  const [feedbackMsg, setFeedbackMsg] = useState('Incorrect decision.');
   128→  const [feedbackAction, setFeedbackAction] = useState('');
   129→  const [busy, setBusy] = useState(false);
   130→  const [previewAsTerminal, setPreviewAsTerminal] = useState<boolean | null>(null); // null = not yet hydrated

   139→  const togglePreviewMode = (on: boolean) => {

   161→  }, [taskId]);

   163→  const taskEvents = useMemo(() => {

   170→

   172→    try {

   174→      setDetail(d);
   175→      setDetailError(null);
   176→    } catch (e) {
   177→      setDetailError(e instanceof ApiClientError ? e.message : 'Task unavailable');

   180→
   181→  const loadSide = useCallback(async () => {

   188→    void loadDetail();

   191→
   192→  useEffect(() => {
   193→    if (!isActive) return;

   209→      toast.error('Stop failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });
   210→    } finally {
   211→      setStopping(false);
   212→    }
   213→  };
   214→

   220→      toast.error('Invalid JSON payload');

   222→    }

   231→      setBusy(false);
   232→    }
   233→  };

   238→      return;
   239→    }

   241→    try {
   242→      await sendTaskFeedback(taskId, {
   243→        message: feedbackMsg.trim(),
   244→        ...(feedbackAction.trim() ? { correctAction: feedbackAction.trim() } : {}),

   248→    } catch (e) {
   249→      toast.error('Feedback failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });

   251→      setBusy(false);
   252→    }
   253→  };

   255→  if (detailError && detail === null) {

   264→  }

   272→          <Skeleton className="h-40 w-full" />
   273→        </div>
   274→      </div>
   275→    );
   276→  }
   277→
   278→  const state = detail.state;

   298→        <>
   299→          <p className="mt-2 text-sm font-medium text-foreground">{state.activeSubgoal.title}</p>

   395→

   435→      {/* Header card — task, status, controls (always visible) */}

   454→              <Badge variant="outline" className={cn('font-mono text-[10px]', detail.mode === 'live' ? 'border-sky-400/30 bg-sky-400/10 text-sky-300' : 'border-white/[0.09] text-muted-foreground')}>
