   116→   116→  const [busy, setBusy] = useState(false);
   117→   117→
   118→   118→  const { events: streamEvents } = useNexoolStream({ taskId, max: 300 });

   128→   128→        /* honest: timeline will simply show live events only */

   139→   139→  }, [backfill, streamEvents]);
   140→   140→
   141→   141→  const isActive = detail ? ACTIVE_STATUSES.has(detail.status) : true;

   147→   147→      setDetailError(null);
   148→   148→    } catch (e) {
   149→   149→      setDetailError(e instanceof ApiClientError ? e.message : 'Task unavailable');
   150→   150→    }
   151→   151→  }, [taskId]);

   153→   153→  const loadSide = useCallback(async () => {

   155→   155→    if (ex.status === 'fulfilled') setExecutions(ex.value);
   156→   156→    if (ctx.status === 'fulfilled') setContext(ctx.value);

   159→   159→  useEffect(() => {

   161→   161→    void loadSide();

   180→   180→    } catch (e) {
   181→   181→      toast.error('Stop failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });
   182→   182→    } finally {
   183→   183→      setStopping(false);
   184→   184→    }
   185→   185→  };

   201→   201→      toast.error('Event injection failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });
   202→   202→    } finally {
   203→   203→      setBusy(false);
   204→   204→    }
   205→   205→  };

   224→   224→    }

   248→   248→  }
   249→   249→
   250→   250→  const state = detail.state;
   251→   251→  const duration = detail.durationMs ?? (detail.startedAt ? Date.now() - new Date(detail.startedAt).getTime() : undefined);
