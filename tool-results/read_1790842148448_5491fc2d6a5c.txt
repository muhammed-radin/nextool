    38→function PlanIcon({ step }: { step: PlanStep }) {
    39→  if (step.status === 'in_progress') return <Loader2 className="size-3.5 animate-spin text-sky-300" aria-hidden />;
    40→  if (step.status === 'completed') return <CheckCircle2 className="size-3.5 text-emerald-400" aria-hidden />;
    41→  if (step.status === 'failed') return <XCircle className="size-3.5 text-rose-400" aria-hidden />;
    42→  return <Circle className="size-3.5 text-slate-600" aria-hidden />;
    43→}

    48→  }
    49→  if (executions.length === 0) return <EmptyState title="No tool calls yet" hint="Executions appear as the CoreModule dispatches tools." />;
    50→  return (
    51→    <div className="space-y-2">
    52→      {executions.map((ex) => (
    53→        <div key={ex.executionId} className="glass-card rounded-md px-3 py-2">

    56→            <span className="font-mono text-xs font-semibold text-foreground">{ex.tool}</span>
    57→            <span className="ml-auto font-mono text-[10px] text-muted-foreground">

    99→export default function TaskPreviewView({ taskId }: { taskId: string }) {
   100→  const setActiveView = useConsoleStore((s) => s.setActiveView);
   101→
   102→  const [detail, setDetail] = useState<TaskDetail | null>(null);
   103→  const [detailError, setDetailError] = useState<string | null>(null);
   104→  const [executions, setExecutions] = useState<ToolExecution[] | null>(null);
   105→  const [context, setContext] = useState<ContextComposition | null>(null);
   106→  const [backfill, setBackfill] = useState<NexToolEvent[]>([]);
   107→
   108→  const [stopOpen, setStopOpen] = useState(false);
   109→  const [stopping, setStopping] = useState(false);
   110→  const [eventOpen, setEventOpen] = useState(false);
   111→  const [eventType, setEventType] = useState('user.message');
   112→  const [eventPayload, setEventPayload] = useState('{}');
   113→  const [feedbackOpen, setFeedbackOpen] = useState(false);
   114→  const [feedbackMsg, setFeedbackMsg] = useState('Incorrect decision.');
   115→  const [feedbackAction, setFeedbackAction] = useState('');
   116→  const [busy, setBusy] = useState(false);
   117→
   118→  const { events: streamEvents } = useNexoolStream({ taskId, max: 300 });
   119→
   120→  // Backfill historical events from REST, then merge with live SSE (dedupe by id).
   121→  useEffect(() => {
   122→    let alive = true;
   123→    getTaskEvents(taskId, { limit: 200 })
   124→      .then((d) => {
   125→        if (alive) setBackfill(d);
   126→      })
   127→      .catch(() => {
   128→        /* honest: timeline will simply show live events only */
   129→      });
   130→    return () => {
   131→      alive = false;
   132→    };
   133→  }, [taskId]);
   134→
   135→  const taskEvents = useMemo(() => {
   136→    const byId = new Map<string, NexToolEvent>();
   137→    for (const ev of [...backfill, ...streamEvents]) byId.set(ev.id, ev);
   138→    return [...byId.values()].sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   139→  }, [backfill, streamEvents]);
   140→
   141→  const isActive = detail ? ACTIVE_STATUSES.has(detail.status) : true;
   142→
   143→  const loadDetail = useCallback(async () => {
   144→    try {
   145→      const d = await getTaskDetail(taskId);
   146→      setDetail(d);
   147→      setDetailError(null);
   148→    } catch (e) {
   149→      setDetailError(e instanceof ApiClientError ? e.message : 'Task unavailable');
   150→    }
   151→  }, [taskId]);
   152→
   153→  const loadSide = useCallback(async () => {
   154→    const [ex, ctx] = await Promise.allSettled([getTaskExecutions(taskId), getTaskContext(taskId)]);
   155→    if (ex.status === 'fulfilled') setExecutions(ex.value);
   156→    if (ctx.status === 'fulfilled') setContext(ctx.value);
   157→  }, [taskId]);
   158→
   159→  useEffect(() => {
   160→    void loadDetail();
   161→    void loadSide();
   162→  }, [loadDetail, loadSide]);
   163→
   164→  useEffect(() => {
   165→    if (!isActive) return;
   166→    const t = setInterval(() => {
   167→      void loadDetail();
   168→      void loadSide();
   169→    }, 2500);
   170→    return () => clearInterval(t);
   171→  }, [isActive, loadDetail, loadSide]);
   172→
   173→  const doStop = async () => {
   174→    setStopping(true);
   175→    try {
   176→      await stopTask(taskId);
   177→      toast.success('Stop signal sent', { description: `Task #${taskId.slice(0, 8)} cancellation requested.` });
   178→      setStopOpen(false);
   179→      void loadDetail();
   180→    } catch (e) {
   181→      toast.error('Stop failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });
   182→    } finally {
   183→      setStopping(false);
   184→    }
   185→  };
   186→
   187→  const doSendEvent = async () => {
   188→    let payload: unknown;
   189→    try {
   190→      payload = eventPayload.trim() ? JSON.parse(eventPayload) : {};
   191→    } catch {
   192→      toast.error('Invalid JSON payload');
   193→      return;
   194→    }
   195→    setBusy(true);
   196→    try {
   197→      await sendTaskEvent(taskId, { type: eventType, payload: payload as Record<string, unknown> });
   198→      toast.success('Event injected', { description: `${eventType} → task #${taskId.slice(0, 8)}` });
   199→      setEventOpen(false);
   200→    } catch (e) {
   201→      toast.error('Event injection failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });
   202→    } finally {
   203→      setBusy(false);
   204→    }
   205→  };
   206→
   207→  const doSendFeedback = async () => {
   208→    if (!feedbackMsg.trim()) {
   209→      toast.error('Feedback message required');
   210→      return;
   211→    }
   212→    setBusy(true);
   213→    try {
   214→      await sendTaskFeedback(taskId, {
   215→        message: feedbackMsg.trim(),
   216→        ...(feedbackAction.trim() ? { correctAction: feedbackAction.trim() } : {}),
   217→      });
   218→      toast.success('Feedback delivered', { description: 'The runtime emitted a user.feedback event.' });
   219→      setFeedbackOpen(false);
   220→    } catch (e) {
   221→      toast.error('Feedback failed', { description: e instanceof ApiClientError ? e.message : 'Unknown error' });
   222→    } finally {
   223→      setBusy(false);
   224→    }
   225→  };
   226→
   227→  if (detailError && detail === null) {
   228→    return (
   229→      <div className="space-y-4">
   230→        <Button variant="ghost" size="sm" className="min-h-11 text-muted-foreground hover:text-foreground" onClick={() => setActiveView('dashboard')}>
   231→          ← Back to dashboard
   232→        </Button>
   233→        <ErrorCard title="Task unavailable" message={detailError} onRetry={loadDetail} />
   234→      </div>
   235→    );
   236→  }
   237→
   238→  if (detail === null) {
   239→    return (
   240→      <div className="space-y-4">
   241→        <Skeleton className="h-28 w-full" />
   242→        <div className="grid gap-4 lg:grid-cols-2">
   243→          <Skeleton className="h-40 w-full" />
   244→          <Skeleton className="h-40 w-full" />
   245→        </div>
   246→      </div>
   247→    );
   248→  }
   249→
   250→  const state = detail.state;
   251→  const duration = detail.durationMs ?? (detail.startedAt ? Date.now() - new Date(detail.startedAt).getTime() : undefined);
   252→  const planSteps = detail.plan ?? state?.plan ?? [];
   253→
   254→  // ---- shared section fragments (rendered in mobile tabs AND desktop columns) ----
   255→
   256→  const goalCard = (
   257→    <section aria-label="Current goal" className="glass-card rounded-lg p-4">
   258→      <SectionTitle icon={<Flag className="size-4 text-sky-300" aria-hidden />} title="Current goal" />
   259→      <p className="mt-2 text-sm text-foreground/90">{state?.goal || '—'}</p>
   260→      <p className="mt-1 font-mono text-[10px] text-muted-foreground">
   261→        iteration {state?.iterationCount ?? 0} · tool calls {state?.toolCallCount ?? 0}
   262→      </p>
   263→    </section>
   264→  );
   265→
   266→  const subgoalCard = (
   267→    <section aria-label="Active subgoal" className="glass-card rounded-lg p-4">
   268→      <SectionTitle icon={<CornerDownRight className="size-4 text-sky-300" aria-hidden />} title="Active subgoal" />
   269→      {state?.activeSubgoal ? (
   270→        <>
   271→          <p className="mt-2 text-sm font-medium text-foreground">{state.activeSubgoal.title}</p>
   272→          <p className="mt-1 text-xs text-muted-foreground">{state.activeSubgoal.reason}</p>
   273→        </>
   274→      ) : (
   275→        <p className="mt-2 text-xs text-muted-foreground/70">No active subgoal — {state?.subgoals?.length ?? 0} subgoal(s) so far.</p>
   276→      )}
   277→    </section>
   278→  );
   279→
   280→  const planSection = (
   281→    <section aria-label="Plan" className="glass-panel rounded-lg p-4">
   282→      <SectionTitle icon={<Play className="size-4 text-sky-300" aria-hidden />} title="Plan" desc="Steps produced by the planner." />
   283→      <div className="mt-3">
   284→        {planSteps.length === 0 ? (
   285→          <EmptyState title="No plan yet" hint="The planner publishes steps once the task starts executing." />
   286→        ) : (
   287→          <ol className="nextool-scroll max-h-64 space-y-1.5 overflow-y-auto pr-1">
   288→            {planSteps.map((step) => (
   289→              <li key={step.id} className="glass-card flex items-start gap-2 rounded-md px-3 py-2">
   290→                <span className="mt-0.5"><PlanIcon step={step} /></span>
   291→                <span className="min-w-0 flex-1">
   292→                  <span className="block text-xs text-foreground/90">{step.title}</span>
   293→                  {step.detail ? <span className="block text-[11px] text-muted-foreground">{step.detail}</span> : null}
   294→                </span>
   295→                <Badge variant="outline" className="border-white/[0.09] font-mono text-[9px] uppercase text-muted-foreground">{step.kind}</Badge>
   296→              </li>
   297→            ))}
   298→          </ol>
   299→        )}
   300→      </div>
   301→    </section>
   302→  );
   303→
   304→  const toolsSection = (
   305→    <section aria-label="Tool calls" className="glass-panel rounded-lg p-4">
   306→      <SectionTitle icon={<Wrench className="size-4 text-sky-300" aria-hidden />} title="Tool calls" desc="Executions with params and results." />
   307→      <div className="mt-3"><ExecutionsPanel executions={executions} /></div>
   308→    </section>
   309→  );
   310→
   311→  const stateSection = (
   312→    <section aria-label="Task state" className="glass-panel rounded-lg p-4">
   313→      <SectionTitle icon={<Braces className="size-4 text-sky-300" aria-hidden />} title="State" desc="MainState snapshot (live JSON)." />
   314→      <div className="mt-3">
   315→        <JsonBlock value={state ?? {}} maxHeight="max-h-80" />
   316→      </div>
   317→    </section>
   318→  );
   319→
   320→  const contextSection = (
   321→    <section aria-label="Context composition" className="glass-panel rounded-lg p-4">
   322→      <SectionTitle icon={<Layers className="size-4 text-sky-300" aria-hidden />} title="Context" desc="Previous + delta + observation + memory + history (assembled by the runtime)." />
   323→      <div className="mt-3 grid gap-3 sm:grid-cols-2 xl:grid-cols-3">
   324→        <ContextPanel title="previous context" value={context?.previousContext ?? null} />
   325→        <ContextPanel title="delta" value={context?.delta ?? null} />
   326→        <ContextPanel title="new observation" value={context?.observation ?? null} />
   327→        <ContextPanel title="memory refs" value={context?.memory ?? null} count={context?.memory?.length} />
   328→        <ContextPanel title="history refs" value={context?.history ?? null} count={context?.history?.length} />
   329→        <div className="glass-card rounded-md p-3">
   330→          <p className="mb-1.5"><TechLabel className="text-[9px]">assembled at</TechLabel></p>
   331→          <p className="font-mono text-xs text-foreground/90">{context ? <TimeAgo iso={context.assembledAt} /> : '—'}</p>
   332→        </div>
   333→      </div>
   334→    </section>
   335→  );
   336→
   337→  const timelineSection = (
   338→    <section aria-label="Event timeline" className="glass-panel rounded-lg p-4">
   339→      <SectionTitle icon={<Radio className="size-4 text-sky-300" aria-hidden />} title="Events timeline" desc="This task's events — REST backfill + live stream (newest last)." />
   340→      <div className="mt-3">
   341→        {taskEvents.length === 0 ? (
   342→          <EmptyState title="No events for this task yet" hint="Runtime events appear here as the task executes." />
   343→        ) : (
   344→          <ol className="nextool-scroll relative max-h-96 space-y-2 overflow-y-auto border-l border-white/[0.08] pl-4 pr-1">
   345→            {taskEvents.map((ev) => (
   346→              <li key={ev.id} className="relative">
   347→                <span
   348→                  aria-hidden
   349→                  className={cn('absolute -left-[21px] top-1.5 size-2 rounded-full ring-2 ring-card', SOURCE_COLORS[ev.source] ?? 'bg-slate-600')}
   350→                />
   351→                <div className="flex flex-wrap items-center gap-1.5">
   352→                  <span className="font-mono text-[10px] text-muted-foreground">[{fmtClock(ev.createdAt)}]</span>
   353→                  <span className="rounded border border-white/[0.09] px-1 font-mono text-[9px] uppercase text-slate-300">{ev.source}</span>
   354→                  <span className="rounded border border-white/[0.09] px-1 font-mono text-[9px] text-cyan-300/90">{ev.type}</span>
   355→                  <span className="min-w-0 flex-1 truncate text-xs text-foreground/85">{ev.message}</span>
   356→                </div>
   357→                {ev.data ? <JsonBlock value={ev.data} maxHeight="max-h-32" className="mt-1" /> : null}
   358→              </li>
   359→            ))}
   360→          </ol>
   361→        )}
   362→      </div>
   363→    </section>
   364→  );
   365→
   366→  const terminal = <RuntimeTerminal taskId={taskId} events={taskEvents} />;
   367→
   368→  return (
   369→    <div className="space-y-4">
   370→      <Button variant="ghost" size="sm" className="min-h-11 text-muted-foreground hover:text-foreground" onClick={() => setActiveView('dashboard')}>
   371→        ← Back to dashboard
   372→      </Button>
   373→
   374→      {/* Header card — task, status, controls (always visible) */}
   375→      <section aria-label="Task overview" className="glass-panel rounded-lg p-4 md:p-6">
   376→        <div className="flex flex-col gap-3 lg:flex-row lg:items-start lg:justify-between">
   377→          <div className="min-w-0 flex-1">
   378→            <p className="break-all font-mono text-xs text-foreground/90">{detail.request}</p>
   379→            <div className="mt-2 flex flex-wrap items-center gap-1.5">
   380→              <StatusChip status={detail.status} />
   381→              <Badge variant="outline" className={cn('font-mono text-[10px]', detail.mode === 'live' ? 'border-sky-400/30 bg-sky-400/10 text-sky-300' : 'border-white/[0.09] text-muted-foreground')}>
   382→                {detail.mode}
   383→              </Badge>
   384→              <Badge variant="outline" className="border-white/[0.09] font-mono text-[10px] text-muted-foreground">L{detail.reasoningLevel}</Badge>
   385→              {detail.sessionId ? <span className="font-mono text-[10px] text-muted-foreground">session {detail.sessionId.slice(0, 12)}</span> : null}
   386→              <span className="font-mono text-[10px] text-muted-foreground">#{detail.id.slice(0, 8)}</span>
   387→            </div>
   388→            <p className="mt-1.5 font-mono text-[10px] text-muted-foreground">
   389→              {detail.steps} steps · {detail.toolCalls} tools{duration !== undefined ? ` · ${fmtMs(duration)}` : ''}
   390→            </p>
   391→          </div>
   392→          <div className="grid grid-cols-2 gap-2 sm:flex sm:flex-wrap sm:items-center">
   393→            <Button
   394→              variant="outline"
   395→              size="sm"
   396→              className="min-h-11 justify-center border-white/[0.09] bg-white/[0.04] text-slate-200 hover:bg-white/[0.08]"
   397→              onClick={() => setEventOpen(true)}
   398→            >
   399→              <Send className="size-3.5" aria-hidden /> Send event
   400→            </Button>
   401→            <Button
   402→              variant="outline"
   403→              size="sm"
   404→              className="min-h-11 justify-center border-white/[0.09] bg-white/[0.04] text-slate-200 hover:bg-white/[0.08]"
   405→              onClick={() => setFeedbackOpen(true)}
   406→            >
   407→              <MessageSquareWarning className="size-3.5" aria-hidden /> Feedback
   408→            </Button>
   409→            {ACTIVE_STATUSES.has(detail.status) ? (
   410→              <Button
   411→                variant="outline"
   412→                size="sm"
   413→                className="col-span-2 min-h-11 justify-center border-rose-500/40 text-rose-300 hover:bg-rose-500/10 sm:col-span-1"
   414→                onClick={() => setStopOpen(true)}
   415→              >
   416→                <Square className="size-3.5" aria-hidden /> Stop task
   417→              </Button>
   418→            ) : null}
   419→          </div>
   420→        </div>
   421→        {detail.statusDetail ? <p className="mt-2 text-xs text-muted-foreground">{detail.statusDetail}</p> : null}
   422→        {detail.error ? <p className="mt-2 font-mono text-xs text-rose-300">{detail.error.code} [{detail.error.stage}]: {detail.error.message}</p> : null}
   423→      </section>
   424→
   425→      {/* Mobile: segmented tabs — Overview · Timeline · Tools · Internals */}
   426→      <div className="lg:hidden">
   427→        <Tabs defaultValue="overview" className="gap-3">
   428→          <TabsList className="glass-card h-auto w-full justify-start gap-1 overflow-x-auto rounded-lg p-1 nextool-scroll" aria-label="Task sections">
   429→            <TabsTrigger
   430→              value="overview"
   431→              className="flex-none min-h-9 px-3 text-xs data-[state=active]:bg-primary-gradient-soft data-[state=active]:shadow-none data-[state=active]:ring-1 data-[state=active]:ring-sky-400/25 dark:data-[state=active]:text-sky-100"
   432→            >
   433→              Overview
   434→            </TabsTrigger>
   435→            <TabsTrigger
