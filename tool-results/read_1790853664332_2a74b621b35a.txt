   240→function StatusBar() {
   241→  const { stats } = useSystemStats();
   242→  const { status } = useGlobalStream();
   243→  const [clock, setClock] = useState<string | null>(null);
   244→
   245→  useEffect(() => {
   246→    const update = () => setClock(new Date().toLocaleTimeString('en-GB', { hour12: false }));
   247→    update();
   248→    const t = setInterval(update, 1000);
   249→    return () => clearInterval(t);
   250→  }, []);
   251→
   252→  const runtimeStatus = stats?.runtimeStatus ?? (status === 'error' || status === 'disconnected' ? 'offline' : 'connecting');
   253→  const tone = statusTone(runtimeStatus);
   254→
   255→  return (
   256→    <footer className="glass-shell mt-auto hidden border-t border-white/[0.07] md:block">
   257→      <div className="mx-auto flex h-9 w-full max-w-[1600px] items-center justify-between gap-3 px-4 font-mono text-[11px] text-slate-400">
   258→        <div className="flex min-w-0 items-center gap-2">
   259→          <span className="font-tech text-[9px] uppercase tracking-widest text-sky-300/70">Runtime</span>
   260→          <span
   261→            aria-label={`Runtime ${runtimeStatus}`}
   262→            className={cn('inline-flex items-center gap-1.5', tone === 'ok' ? 'text-emerald-300' : tone === 'warn' ? 'text-amber-300' : 'text-rose-300')}
   263→          >
   264→            <span className={cn('size-1.5 rounded-full', tone === 'ok' ? 'animate-pulse bg-emerald-400' : tone === 'warn' ? 'bg-amber-400' : 'bg-rose-400')} aria-hidden />
   265→            {stats ? runtimeStatus : 'offline'}
   266→          </span>
   267→        </div>
   268→        <div className="hidden min-w-0 items-center gap-3 truncate lg:flex">
   269→          <span>active {stats ? stats.tasks.active : '—'}</span>
   270→          <span>live {stats ? stats.tasks.live : '—'}</span>
   271→          <span>engine {stats ? stats.engine.active : '—'}</span>
   272→          <span>up {stats ? fmtUptime(stats.runtimeUptimeSec) : '—'}</span>
   273→        </div>
   274→        <div className="flex items-center gap-2">
   275→          <span aria-hidden className={cn('size-1.5 rounded-full', status === 'connected' ? 'animate-pulse bg-emerald-400' : status === 'error' ? 'bg-rose-400' : status === 'reconnecting' ? 'bg-amber-400' : 'bg-sky-400')} />

   404→function ToolEditorRoute() {
   405→  const toolEditor = useConsoleStore((s) => s.toolEditor);
   406→  const closeToolEditor = useConsoleStore((s) => s.closeToolEditor);
   407→  // The parent remounts this route per editor session (key=toolEditorKey), so
   408→  // the prefill state can be initialized lazily without effect-time setState.
   409→  const [state, setState] = useState<{ entry: ToolEntry | null; error: string | null }>(() => ({
   410→    entry: toolEditor && toolEditor.mode !== 'edit' ? toolEditor.source ?? null : null,
   411→    error: null,
   412→  }));
   413→  const { entry, error } = state;
   414→
   415→  useEffect(() => {
   416→    if (!toolEditor || toolEditor.mode !== 'edit' || !toolEditor.name) return;
   417→    let alive = true;
   418→    getTool(toolEditor.name)
   419→      .then((d) => {
   420→        if (alive) setState({ entry: d, error: null });
   421→      })
   422→      .catch((e) => {
   423→        if (alive) setState({ entry: null, error: e instanceof ApiClientError ? e.message : 'Tool not found' });
   424→      });
   425→    return () => {
   426→      alive = false;
   427→    };
   428→  }, [toolEditor]);
   429→
   430→  if (!toolEditor) return null;
   431→
   432→  const loading = toolEditor.mode === 'edit' && entry === null && error === null;
   433→
   434→  if (error) {
   435→    return (
   436→      <div className="space-y-4">
   437→        <p role="alert" className="rounded-lg border border-rose-400/30 bg-rose-400/5 p-4 text-sm text-rose-300">{error}</p>
   438→        <Button variant="outline" size="sm" onClick={closeToolEditor}>← Back to tools</Button>
   439→      </div>
   440→    );
   441→  }
   442→  if (loading) {
   443→    return (
   444→      <div className="space-y-4">
   445→        <div className="h-16 w-full animate-pulse rounded-lg bg-white/[0.06]" />
   446→        <div className="h-96 w-full animate-pulse rounded-lg bg-white/[0.05]" />
   447→      </div>
   448→    );
   449→  }
   450→
   451→  const initial = toolEditor.mode === 'edit'
   452→    ? entry
   453→    : entry
   454→      ? {
   455→          ...entry,
   456→          name: entry.name.includes('.') ? `${entry.name.split('.')[0]}.copy` : `${entry.name}.copy`,
   457→          toolVersion: '1.0.0',
   458→          stats: { callCount: 0, successCount: 0, failureCount: 0, timeoutCount: 0, avgMs: 0, enabled: true },
   459→        }
   460→      : null;
   461→
   462→  return (
   463→    <ToolEditorView
   464→      toolName={toolEditor.mode === 'edit' ? toolEditor.name : null}
   465→      initial={initial}
   466→      onSaved={() => closeToolEditor()}

   473→function ViewRouter() {
   474→  const activeView = useConsoleStore((s) => s.activeView);
   475→  const selectedTaskId = useConsoleStore((s) => s.selectedTaskId);
   476→  const openToolEditor = useConsoleStore((s) => s.openToolEditor);
   477→  const toolEditorKey = useConsoleStore((s) => s.toolEditorKey);
   478→
   479→  const view = (() => {
   480→    switch (activeView) {
   481→      case 'dashboard':
   482→        return <DashboardView />;
   483→      case 'task-console':
   484→        return <TaskConsoleView />;
   485→      case 'live-monitor':
   486→        return <LiveMonitorView />;
   487→      case 'tools':
   488→        return <ToolsView onOpenEditor={openToolEditor} />;
   489→      case 'tool-editor':
   490→        return <ToolEditorRoute key={toolEditorKey} />;
   491→      case 'training':
   492→        return <TrainingView />;
   493→      case 'benchmark':
   494→        return <BenchmarkView />;
   495→      case 'memory':
   496→        return <MemoryView />;
   497→      case 'live-state':
   498→        return <LiveStateView />;
   499→      case 'events':
   500→        return <EventsView />;
   501→      case 'history':
   502→        return <HistoryView />;
   503→      case 'models':
   504→        return <ModelsView />;
   505→      case 'datasets':
   506→        return <DatasetsView />;
   507→      case 'docs':
   508→        return <DocsView />;
   509→      case 'settings':
   510→        return <SettingsView />;
   511→      case 'task-preview':
   512→        return selectedTaskId ? <TaskPreviewView key={selectedTaskId} taskId={selectedTaskId} /> : <DashboardView />;
   513→      default:
   514→        return <DashboardView />;
   515→    }
   516→  })();
   517→
   518→  return (
   519→    <AnimatePresence mode="wait" initial={false}>
   520→      <motion.div
   521→        key={activeView}
   522→        initial={{ opacity: 0, y: 8 }}
   523→        animate={{ opacity: 1, y: 0 }}
   524→        exit={{ opacity: 0, y: -8 }}
   525→        transition={{ duration: 0.16, ease: 'easeOut' }}
   526→      >
   527→        {view}
   528→      </motion.div>
   529→    </AnimatePresence>
   530→  );
   531→}
