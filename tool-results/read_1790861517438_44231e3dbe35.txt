   242→function StatusBar() {
   243→  const { stats } = useSystemStats();
   244→  const { status } = useGlobalStream();
   245→  const [clock, setClock] = useState<string | null>(null);
   246→
   247→  useEffect(() => {
   248→    const update = () => setClock(new Date().toLocaleTimeString('en-GB', { hour12: false }));
   249→    update();
   250→    const t = setInterval(update, 1000);
   251→    return () => clearInterval(t);
   252→  }, []);
   253→
   254→  const runtimeStatus = stats?.runtimeStatus ?? (status === 'error' || status === 'disconnected' ? 'offline' : 'connecting');
   255→  const tone = statusTone(runtimeStatus);
   256→
   257→  return (
   258→    <footer className="glass-shell mt-auto hidden border-t border-white/[0.07] md:block">
   259→      <div className="mx-auto flex h-9 w-full max-w-[1600px] items-center justify-between gap-3 px-4 font-mono text-[11px] text-slate-400">

   261→          <span className="font-tech text-[9px] uppercase tracking-widest text-sky-300/70">Runtime</span>
   262→          <span
   263→            aria-label={`Runtime ${runtimeStatus}`}
   264→            className={cn('inline-flex items-center gap-1.5', tone === 'ok' ? 'text-emerald-300' : tone === 'warn' ? 'text-amber-300' : 'text-rose-300')}
   265→          >
   266→            <span className={cn('size-1.5 rounded-full', tone === 'ok' ? 'animate-pulse bg-emerald-400' : tone === 'warn' ? 'bg-amber-400' : 'bg-rose-400')} aria-hidden />

   406→function ToolEditorRoute() {
   407→  const toolEditor = useConsoleStore((s) => s.toolEditor);
   408→  const closeToolEditor = useConsoleStore((s) => s.closeToolEditor);
   409→  // The parent remounts this route per editor session (key=toolEditorKey), so
   410→  // the prefill state can be initialized lazily without effect-time setState.
   411→  const [state, setState] = useState<{ entry: ToolEntry | null; error: string | null }>(() => ({
   412→    entry: toolEditor && toolEditor.mode !== 'edit' ? toolEditor.source ?? null : null,
   413→    error: null,
   414→  }));
   415→  const { entry, error } = state;
   416→
   417→  useEffect(() => {
   418→    if (!toolEditor || toolEditor.mode !== 'edit' || !toolEditor.name) return;
   419→    let alive = true;
   420→    getTool(toolEditor.name)
   421→      .then((d) => {
   422→        if (alive) setState({ entry: d, error: null });
   423→      })
   424→      .catch((e) => {
   425→        if (alive) setState({ entry: null, error: e instanceof ApiClientError ? e.message : 'Tool not found' });
   426→      });
   427→    return () => {
   428→      alive = false;
   429→    };
   430→  }, [toolEditor]);
   431→
   432→  if (!toolEditor) return null;
   433→
   434→  const loading = toolEditor.mode === 'edit' && entry === null && error === null;
   435→
   436→  if (error) {
   437→    return (
   438→      <div className="space-y-4">
   439→        <p role="alert" className="rounded-lg border border-rose-400/30 bg-rose-400/5 p-4 text-sm text-rose-300">{error}</p>
   440→        <Button variant="outline" size="sm" onClick={closeToolEditor}>← Back to tools</Button>
   441→      </div>
   442→    );
   443→  }
   444→  if (loading) {
   445→    return (
   446→      <div className="space-y-4">
   447→        <div className="h-16 w-full animate-pulse rounded-lg bg-white/[0.06]" />
   448→        <div className="h-96 w-full animate-pulse rounded-lg bg-white/[0.05]" />
   449→      </div>
   450→    );
   451→  }
   452→
   453→  const initial = toolEditor.mode === 'edit'
   454→    ? entry
   455→    : entry
   456→      ? {
   457→          ...entry,
   458→          name: entry.name.includes('.') ? `${entry.name.split('.')[0]}.copy` : `${entry.name}.copy`,
   459→          toolVersion: '1.0.0',
   460→          stats: { callCount: 0, successCount: 0, failureCount: 0, timeoutCount: 0, avgMs: 0, enabled: true },
   461→        }
   462→      : null;
   463→
   464→  return (

   475→function ViewRouter() {
   476→  const activeView = useConsoleStore((s) => s.activeView);
   477→  const selectedTaskId = useConsoleStore((s) => s.selectedTaskId);
   478→  const openToolEditor = useConsoleStore((s) => s.openToolEditor);
   479→  const toolEditorKey = useConsoleStore((s) => s.toolEditorKey);
   480→
   481→  const view = (() => {
   482→    switch (activeView) {
   483→      case 'dashboard':
   484→        return <DashboardView />;
   485→      case 'task-console':
   486→        return <TaskConsoleView />;
   487→      case 'live-monitor':
   488→        return <LiveMonitorView />;
   489→      case 'tools':
   490→        return <ToolsView onOpenEditor={openToolEditor} />;
   491→      case 'tool-editor':
   492→        return <ToolEditorRoute key={toolEditorKey} />;
   493→      case 'training':
   494→        return <TrainingView />;
   495→      case 'benchmark':
   496→        return <BenchmarkView />;
   497→      case 'memory':
   498→        return <MemoryView />;
   499→      case 'live-state':
   500→        return <LiveStateView />;
   501→      case 'events':
   502→        return <EventsView />;
   503→      case 'history':
   504→        return <HistoryView />;
   505→      case 'models':
   506→        return <ModelsView />;
   507→      case 'datasets':
   508→        return <DatasetsView />;
   509→      case 'docs':
   510→        return <DocsView />;
   511→      case 'settings':
   512→        return <SettingsView />;
   513→      case 'task-preview':
   514→        return selectedTaskId ? <TaskPreviewView key={selectedTaskId} taskId={selectedTaskId} /> : <DashboardView />;
   515→      default:
   516→        return <DashboardView />;
   517→    }
   518→  })();
   519→
   520→  return (
   521→    <AnimatePresence mode="wait" initial={false}>
   522→      <motion.div
   523→        key={activeView}
   524→        initial={{ opacity: 0, y: 8 }}
   525→        animate={{ opacity: 1, y: 0 }}
   526→        exit={{ opacity: 0, y: -8 }}
   527→        transition={{ duration: 0.16, ease: 'easeOut' }}
   528→      >
   529→        {view}
   530→      </motion.div>
   531→    </AnimatePresence>
   532→  );
   533→}
